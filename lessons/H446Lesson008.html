Store the possible digits in order: "0123456789ABCDEF". A remainder becomes a position in this string.

Take the first remainder

189 % 16 is 13. digits.Substring(13, 1) is "D", the right-hand digit.

Divide and repeat

189 / 16 is 11. Then 11 % 16 is 11, so prepend "B" to make "BD".

Stop when the quotient is zero

The loop ends. The output is BD: 11 × 16 + 13 = 189.

Convert denary 189 to upper-case hexadecimal, beginning with the text "189". [3 marks]

Full .NET Fiddle Console model

using System;

public class Program
{
    public static string DenaryToHex(int denary)
    {
        string digits = "0123456789ABCDEF";
        string hexadecimal = "";

        while (denary > 0)
        {
            int remainder = denary % 16;
            hexadecimal = digits.Substring(remainder, 1) + hexadecimal;
            denary = denary / 16;
        }

        return hexadecimal;
    }

    public static void Main()
    {
        string denaryText = "189";
        int denary = int.Parse(denaryText);
        Console.WriteLine(DenaryToHex(denary));
    }
}

Output: BD

Why it works: it changes representations, not the value: "189" → 189 → "BD".

Build both operations [6]

H446 coding extension — applies the conversion and arithmetic methods above
Write a complete program that outputs labelled 8-bit sum and difference. [6 marks]
using System;

public class Program
{
    public static int BinaryToDenary(string binaryText)
    {
        int denary = 0;
        for (int position = 0; position < binaryText.Length; position++)
        {
            int bit = int.Parse(binaryText.Substring(position, 1));
            denary = denary * 2 + bit;
        }
        return denary;
    }

    public static string DenaryToBinary8(int denary)
    {
        string binary = "";
        for (int position = 0; position < 8; position++)
        {
            if (denary % 2 == 0)
            {
                binary = "0" + binary;
            }
            else
            {
                binary = "1" + binary;
            }
            denary = denary / 2;
        }
        return binary;
    }

    public static void Main()
    {
        string first = "00110111";
        string second = "00001101";
        int firstValue = BinaryToDenary(first);
        int secondValue = BinaryToDenary(second);
        Console.WriteLine("Sum: " + DenaryToBinary8(firstValue + secondValue));
        Console.WriteLine("Difference: " + DenaryToBinary8(firstValue - secondValue));
    }
}

Sum: 01000100 Difference: 00101010

Quotient
