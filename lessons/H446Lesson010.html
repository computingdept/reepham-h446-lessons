Convert 10011101 0110 into denary. 8-bit mantissa and 4-bit exponent; both two’s complement. [3 marks]

Full .NET Fiddle Console model

using System;

public class Program
{
    public static int PowerOfTwo(int exponent)
    {
        int value = 1;
        for (int count = 0; count < exponent; count++)
        {
            value = value * 2;
        }
        return value;
    }

    public static int TwosComplementToDenary(string bits)
    {
        int value = 0;
        for (int position = 0; position < bits.Length; position++)
        {
            int bit = int.Parse(bits.Substring(position, 1));
            value = value * 2 + bit;
        }
        if (bits.Substring(0, 1) == "1")
        {
            value = value - PowerOfTwo(bits.Length);
        }
        return value;
    }

    public static double MantissaToDenary(string mantissa)
    {
        double value = -int.Parse(mantissa.Substring(0, 1));
        double weight = 0.5;
        for (int position = 1; position < mantissa.Length; position++)
        {
            int bit = int.Parse(mantissa.Substring(position, 1));
            value = value + bit * weight;
            weight = weight / 2;
        }
        return value;
    }

    public static double ApplyExponent(double value, int exponent)
    {
        while (exponent > 0)
        {
            value = value * 2;
            exponent = exponent - 1;
        }
        while (exponent < 0)
        {
            value = value / 2;
            exponent = exponent + 1;
        }
        return value;
    }

    public static void Main()
    {
        string mantissa = "10011101";
        int exponent = TwosComplementToDenary("0110");
        double value = ApplyExponent(MantissaToDenary(mantissa), exponent);
        Console.WriteLine(value);
    }
}

Output: -49.5

Exam answer: −49.5. The code represents the same two stages as the mark scheme: decode the mantissa, then shift by the exponent.

Normalisation rule

Positive mantissa: begin 01.

Negative mantissa: begin 10.

Every left shift of the mantissa requires the exponent to decrease by 1.

Why?

Left shifting doubles the mantissa, so halving the scale factor keeps the final value unchanged.

using System;

public class Program
{
    public static int PowerOfTwo(int exponent)
    {
        int value = 1;
        for (int count = 0; count < exponent; count++)
        {
            value = value * 2;
        }
        return value;
    }

    public static int TwosComplementToDenary(string bits)
    {
        int value = 0;
        for (int position = 0; position < bits.Length; position++)
        {
            int bit = int.Parse(bits.Substring(position, 1));
            value = value * 2 + bit;
        }
        if (bits.Substring(0, 1) == "1")
        {
            value = value - PowerOfTwo(bits.Length);
        }
        return value;
    }

    public static string DenaryToTwosComplement(int value, int width)
    {
        if (value < 0) value = value + PowerOfTwo(width);
        string bits = "";
        for (int position = 0; position < width; position++)
        {
            if (value % 2 == 0) bits = "0" + bits;
            else bits = "1" + bits;
            value = value / 2;
        }
        return bits;
    }

    public static bool IsNormalised(string mantissa)
    {
        string firstTwo = mantissa.Substring(0, 2);
        return firstTwo == "01" || firstTwo == "10";
    }

    public static string ShiftMantissaLeft(string mantissa)
    {
        return mantissa.Substring(1, mantissa.Length - 1) + "0";
    }

    public static void Main()
    {
        string mantissa = "00001011";
        int exponent = TwosComplementToDenary("0111");

        while (!IsNormalised(mantissa))
        {
            mantissa = ShiftMantissaLeft(mantissa);
            exponent = exponent - 1;
        }

        Console.WriteLine(mantissa + " " +
            DenaryToTwosComplement(exponent, 4));
    }
}

Precision matters

Say it like a computer scientist

Say the missing term or definition before each click. One answer appears at a time.

Floating point
A representation that stores a mantissa and exponent so the binary point can effectively move.
Mantissa
The signed fractional part that contains the significant binary digits of a floating-point value.
Exponent
The signed value that controls how far the binary point is shifted.
Normalised
A mantissa beginning 01 when positive or 10 when negative.
Binary point
The binary equivalent of a decimal point; it separates whole and fractional places.
Precision
How finely a representation can distinguish nearby values; more mantissa bits improve it.
Range
The span of large and small values a representation can store; more exponent bits increase it.
Shift
Moving digits left or right relative to the binary point, changing the scale of the mantissa.
Underflow
When a value is too small in magnitude to be represented with the available exponent range.
